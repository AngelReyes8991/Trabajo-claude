Actualizar los precios de Planes con una sola sentencia UPDATE con CASE.

SELECT PlanID, NombrePlan, Pantallas, Precio AS PrecioAnterior
FROM Planes
ORDER BY Precio DESC;
GO

-- 2.2) Unica sentencia UPDATE con CASE
UPDATE Planes
SET Precio = CASE
                WHEN Pantallas = 1            THEN Precio * 1.05   -- +5%
                WHEN Pantallas BETWEEN 2 AND 3 THEN Precio * 1.10   -- +10%
                WHEN Pantallas >= 4           THEN Precio * 1.15   -- +15%
                ELSE Precio
             END;
GO

-- 2.3) Planes con sus precios actualizados, de mayor a menor precio

        NombrePlan,
        Pantallas,
        CAST(Precio AS DECIMAL(10,2)) AS PrecioActualizado

ORDER BY Precio DESC;

PREGUNTA 4 (2 puntos)
Crear la tabla ReporteConsumoUsuarios mediante SELECT INTO, solo con usuarios
activos, usando INNER JOIN, COUNT, SUM, AVG y GROUP BY.
===============================================================================

-- 4.1) Si la tabla ya existe se elimina, porque SELECT INTO la crea
IF OBJECT_ID('ReporteConsumoUsuarios', 'U') IS NOT NULL
    DROP TABLE ReporteConsumoUsuarios;
GO

-- 4.2) Creacion de la tabla con SELECT INTO
SELECT  U.UsuarioID,
        U.Nombre,
        U.Pais,
        COUNT(R.ReproduccionID) AS CantidadReproducciones,
        SUM(R.MinutosVistos)    AS TotalMinutosVistos,
        CAST(AVG(CAST(R.MinutosVistos AS DECIMAL(10,2))) AS DECIMAL(10,2))
                                AS PromedioMinutosPorReproduccion
INTO ReporteConsumoUsuarios
FROM         Usuarios       U
INNER JOIN   Reproducciones R ON R.UsuarioID = U.UsuarioID
WHERE U.Estado = 'Activo'
GROUP BY U.UsuarioID, U.Nombre, U.Pais;
GO

-- 4.3) Contenido de la tabla creada
SELECT  UsuarioID,
        Nombre,
        Pais,
        CantidadReproducciones,
        TotalMinutosVistos,
        PromedioMinutosPorReproduccion
FROM ReporteConsumoUsuarios
ORDER BY TotalMinutosVistos DESC;
GO

PREGUNTA 5 (2 puntos)
Dos transacciones sobre la tabla Planes: la primera con ROLLBACK y la segunda
con COMMIT, mostrando los precios antes y despues de cada operacion.
===============================================================================

-------------------------------- TRANSACCION 1 --------------------------------
-- Incremento del 20% a los planes con 2 o mas pantallas y ROLLBACK

-- 5.1) Precios ANTES de la transaccion 1
SELECT 'ANTES DE LA TRANSACCION 1' AS Momento,
       PlanID, NombrePlan, Pantallas, CAST(Precio AS DECIMAL(10,2)) AS Precio
FROM Planes
ORDER BY PlanID;
GO

BEGIN TRANSACTION;

    UPDATE Planes
    SET Precio = Precio * 1.20
    WHERE Pantallas >= 2;

    -- 5.2) Precios DENTRO de la transaccion (cambios aun no confirmados)
    SELECT 'DENTRO DE LA TRANSACCION 1' AS Momento,
           PlanID, NombrePlan, Pantallas, CAST(Precio AS DECIMAL(10,2)) AS Precio
    FROM Planes
    ORDER BY PlanID;

ROLLBACK TRANSACTION;
GO

-- 5.3) Precios DESPUES del ROLLBACK: deben ser iguales a los del paso 5.1
SELECT 'DESPUES DEL ROLLBACK' AS Momento,
       PlanID, NombrePlan, Pantallas, CAST(Precio AS DECIMAL(10,2)) AS Precio
FROM Planes
ORDER BY PlanID;
GO

-------------------------------- TRANSACCION 2 --------------------------------
-- Incremento de 5 unidades monetarias al plan Premium y COMMIT

-- 5.4) Precio ANTES de la transaccion 2
SELECT 'ANTES DE LA TRANSACCION 2' AS Momento,
       PlanID, NombrePlan, Pantallas, CAST(Precio AS DECIMAL(10,2)) AS Precio
FROM Planes
WHERE NombrePlan = 'Premium';
GO

BEGIN TRANSACTION;

    UPDATE Planes
    SET Precio = Precio + 5
    WHERE NombrePlan = 'Premium';

COMMIT TRANSACTION;
GO

-- 5.5) Precio DESPUES del COMMIT: el cambio queda confirmado
SELECT 'DESPUES DEL COMMIT' AS Momento,
       PlanID, NombrePlan, Pantallas, CAST(Precio AS DECIMAL(10,2)) AS Precio
FROM Planes
WHERE NombrePlan = 'Premium';
GO

PREGUNTA 7 (2 puntos)
Vista vw_EstadisticasUsuarios con LEFT JOIN e ISNULL, incluyendo a los usuarios
sin reproducciones con valores en cero.
===============================================================================

CREATE OR ALTER VIEW vw_EstadisticasUsuarios
AS
    SELECT  U.UsuarioID,
            U.Nombre,
            U.Pais,
            U.Estado,
            COUNT(R.ReproduccionID)         AS CantidadReproducciones,
            ISNULL(SUM(R.MinutosVistos), 0) AS TotalMinutosVistos,
            ISNULL(CAST(AVG(CAST(R.MinutosVistos AS DECIMAL(10,2))) AS DECIMAL(10,2)), 0)
                                            AS PromedioMinutosPorReproduccion
    FROM        Usuarios       U
    LEFT JOIN   Reproducciones R ON R.UsuarioID = U.UsuarioID
    GROUP BY U.UsuarioID, U.Nombre, U.Pais, U.Estado;
GO

-- 7.1) Solo usuarios con mas de 100 minutos vistos, de mayor a menor
SELECT  UsuarioID,
        Nombre,
        Pais,
        Estado,
        CantidadReproducciones,
        TotalMinutosVistos,
        PromedioMinutosPorReproduccion
FROM vw_EstadisticasUsuarios
WHERE TotalMinutosVistos > 100
ORDER BY TotalMinutosVistos DESC;
GO

PREGUNTA 8 (2 puntos)
Vista vw_ContenidosSuperioresPromedio: contenidos cuyo total de minutos vistos
supera el promedio de minutos acumulados por contenido con reproducciones.
===============================================================================

CREATE OR ALTER VIEW vw_ContenidosSuperioresPromedio
AS
    SELECT  C.ContenidoID,
            C.Titulo,
            C.Tipo,
            C.Categoria,
            COUNT(R.ReproduccionID) AS CantidadReproducciones,
            SUM(R.MinutosVistos)    AS TotalMinutosVistos
    FROM         Contenidos     C
    INNER JOIN   Reproducciones R ON R.ContenidoID = C.ContenidoID
    GROUP BY C.ContenidoID, C.Titulo, C.Tipo, C.Categoria
    HAVING SUM(R.MinutosVistos) >
    (
        -- Subconsulta: promedio de minutos acumulados por contenido
        -- considerando unicamente los contenidos que tienen reproducciones
        SELECT AVG(CAST(T.TotalMinutos AS DECIMAL(18,2)))
        FROM (
                SELECT ContenidoID, SUM(MinutosVistos) AS TotalMinutos
                FROM Reproducciones
                GROUP BY ContenidoID
             ) AS T
    );
GO

-- 8.1) Consulta de la vista ordenada por total de minutos vistos (desc)
SELECT  ContenidoID,
        Titulo,
        Tipo,
        Categoria,
        CantidadReproducciones,
        TotalMinutosVistos
FROM vw_ContenidosSuperioresPromedio
ORDER BY TotalMinutosVistos DESC;
GO

             AND object_id = OBJECT_ID('Contenidos'))

    DROP INDEX IX_Contenidos_Tipo_Categoria_Anio ON Contenidos;

    PRINT 'El indice IX_Contenidos_Tipo_Categoria_Anio no existe. Se creara.';
GO

-- 9.2) Creacion del indice no agrupado compuesto
CREATE NONCLUSTERED INDEX IX_Contenidos_Tipo_Categoria_Anio
ON Contenidos (Tipo, Categoria, Anio);
GO

        Categoria,
        Anio
